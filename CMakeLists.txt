cmake_minimum_required(VERSION 3.20)
project(GpuSim LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(gpusim src/main.cpp)
target_include_directories(gpusim PRIVATE include)
target_compile_options(gpusim PRIVATE -Wall -Wextra -Wpedantic)
include(CTest)
add_executable(scheduler_tests tests/scheduler_tests.cpp)
target_include_directories(scheduler_tests PRIVATE include)
add_test(NAME scheduler COMMAND scheduler_tests)
option(GPUSIM_CUDA "Build CUDA vector-add profiler" OFF)
if(GPUSIM_CUDA)
 enable_language(CUDA)
 add_executable(cuda_profile cuda/profile.cu)
endif()
